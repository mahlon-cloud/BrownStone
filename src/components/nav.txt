'use client';

import React, { useEffect, useState } from 'react';
import Image from 'next/image';

export default function Celestia() {
    const [isMobile, setIsMobile] = useState(false);

    useEffect(() => {
        const handleResize = () => {
            setIsMobile(window.innerWidth < 768);
        };
        handleResize();
        window.addEventListener('resize', handleResize);
        return () => window.removeEventListener('resize', handleResize);
    }, []);

    return (
        <div className="w-full flex flex-col items-center justify-start bg-white dark:bg-gray-900">
            {/* Full-screen width hero container */}
            <div className="w-screen max-h-200 relative overflow-hidden bg-gradient-to-b from-gray-200 to-gray-100 flex items-center justify-center">
                {/* Image container with responsive sizing */}
                <div className="relative w-full h-200">
                    <img
                        src={isMobile ? '/bkgx.jpg' : '/cele.png'}
                        alt="Celestia Hero"
                        className="w-full object-cover"
                        loading="lazy"
                    />
                    {/* Dark overlay for text readability */}
                    <div className="absolute inset-0 bg-black/30" />
                </div>

                {/* Hero text content */}
                <div className="absolute inset-0 flex flex-col items-center justify-center text-white text-center px-6 z-10">
                    <h1 className="text-3xl md:text-5xl lg:text-7xl font-extrabold mb-4 drop-shadow-lg">
                        Celestia
                    </h1>
                    <p className="text-sm md:text-lg lg:text-2xl max-w-2xl drop-shadow-md">
                        Discover extraordinary living experiences
                    </p>
                </div>
            </div>

            {/* Additional content section */}
            <section className="w-full py-12 md:py-16 lg:py-20 px-6 max-w-6xl mx-auto">
                <div className="text-center">
                    <h2 className="text-2xl md:text-4xl font-bold text-gray-800 mb-4">Welcome to Celestia</h2>
                    <p className="text-gray-600 text-base md:text-lg leading-relaxed">
                        Experience premium properties with cutting-edge design and sustainable living solutions.
                    </p>
                </div>
            </section>
        </div>
    );
}
